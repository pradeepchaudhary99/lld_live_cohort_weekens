// 1. POSITION — Where something is on board
Position {
    row, col
    isValid()
}

// 2. PLAYER — Who is playing
Player {
    id          // "X", "O", "RED", etc.
    getType()   // HUMAN, AI
}

// 3. MOVE — Action player takes
Move {
    from        // starting position (can be null for some games)
    to          // ending position
}

// 4. BOARD — Game state (data)
Board {
    grid[][]                          // 2D array of cells
    
    init()                            // setup board
    executeMove(move)                 // apply move
    undoMove(move)                    // revert move
    getPossibleMoves(player)          // all valid moves
    isGameOver()                      // game ended?
    getWinner()                       // who won?
    display()                         // print board
}

// 5. GAME — Main controller
Game {
    board
    currentPlayer
    
    start()                           // begin game
    makeMove(move)                    // validate + execute
    switchTurn()                      // next player
    isGameOver()
    getWinner()
}

// TIC TAC TOE
/*
Board size: 3x3
Position: (row, col) 0-2
Move: pick empty cell
Win: 3 in row/col/diagonal
Valid: cell must be empty
*/

// CONNECT 4
/*
Board size: 6x7
Position: (row, col)
Move: drop piece in column (gravity)
Win: 4 in row/col/diagonal
Valid: column not full, piece falls
*/

// CHESS
/*
Board size: 8x8
Position: (row, col) 0-7
Move: (from, to)
Win: checkmate
Valid: piece can move, king not in check after move
Special: check detection, multiple pieces with rules
*/

// CHECKERS
/*
Board size: 8x8
Position: (row, col)
Move: diagonal (1 or 2 squares)
Win: capture all pieces or opponent can't move
Valid: diagonal path clear, can't jump own pieces
*/

// SNAKE & LADDER
/*
Board size: 10x10 (100 squares)
Position: single number 1-100
Move: random dice + position
Win: reach 100
Valid: advance by dice value
Special: snake/ladder changes position
*/

/*
1. INTERFACES (if designing for multiple games)
   - Position: where things are
   - Move: what players do
   - Board: state + rules
   - Game: controller

2. SEPARATION OF CONCERNS
   - Board = data + game logic
   - Game = control flow + turn management

3. VALIDATION
   - isValidMove() before executeMove()
   - Simulate move, check if legal, undo if not

4. WIN CONDITIONS
   - Check after each move
   - Can be pattern matching (3 in row)
   - Or state checking (capture all pieces)

5. SCALABILITY
   - Grid-based games: 2D array
   - Position-based games: Map or List
   - Easy to extend to new games with same framework
*/
